def test_exact_metadata_counts_roles_and_missing(q,metadata):
    rows=[metadata['inherited_pair']]+[r['pair'] for r in metadata['pairs']]
    assert len(rows)==len({r['id'] for r in rows})==25
    assert len({r['patient_group'] for r in rows})==13
    assert all(r['role']=='TRAIN' for r in rows)
    assert [(r['patient_group'],r['phase']) for r in metadata['missing_annotations']]==[
      ('RESECT:Case11','during'),('RESECT:Case11','after'),('RESECT:Case15','during')]
    assert not set(r['patient_group'] for r in rows)&set(r['patient_group'] for r in metadata['excluded_members'])
    assert len(q.initial_outcomes(metadata))==25
    assert metadata['rights']['visible_cavity_not_total_removed_tissue_or_reward'] is True
    assert metadata['rights']['commercial_clearance'] is False

@pytest.mark.parametrize('mutation',['raw_summary','geometry_translation','geometry_shape','image_budget'])
def test_raw_header_and_reported_geometry_are_same_record(q,receipt,mutation):
    r,row,i,d=receipt
    if mutation=='raw_summary': r['image_qc']['raw_grid']['raw_grid']['shape']=[9,9,9]
    elif mutation=='geometry_translation':
        for h in (r['image_qc']['geometry_qc']['header'],r['mask_geometry']['header']):
            h['affine_ras_mm'][0][3]=1234.
    elif mutation=='geometry_shape':
        for h in (r['image_qc']['geometry_qc']['header'],r['mask_geometry']['header']):h['shape']=[2,2,3]
        r['image_qc']['scalar_qc']['voxels']=12
        r['mask_qc']['decoding_budget']['voxels']=12
        r['mask_qc']['background_voxels']=11
    elif mutation=='image_budget':r['image_qc']['decoding_budget']['voxels']=999
    r['pair_geometry']=q.pair_geometry(r['image_qc']['geometry_qc']['header'],r['mask_geometry']['header'])
    with pytest.raises(q.Refusal):q.receipt_contract(r,row,i,d)

@pytest.mark.parametrize('status,code',[('timeout',None),('worker_failed',1),('completed',1)])
def test_parent_failure_never_promotes_saved_scientific_receipt(q,metadata,monkeypatch,status,code):
    import tempfile
    with tempfile.TemporaryDirectory(dir=ROOT/'build') as directory:
        run=Path(directory);row=metadata['pairs'][0]
        declaration={'run_id':run.name}
        monkeypatch.setattr(q,'validate_execution',lambda *a,**k:None)
        monkeypatch.setattr(q,'validate_inputs',lambda *a,**k:None)
        def supervisor(command,log,*,deadline,on_start):
            on_start(987654321)
            trial=run/'pairs'/row['pair']['id'];intent=json.loads((trial/'intent.json').read_bytes())
            r=q.base_receipt(row,intent,declaration)
            r.update(status='review_failed',fixity_before='passed',fixity_after='passed')
            q.save(trial/'receipt.json',r)
            return status,code
        monkeypatch.setattr(q,'supervise',supervisor)
        outcome={'status':'not_attempted'}
        assert q.run_pair(run,row,metadata,declaration,outcome,time.monotonic()+60) is False
        assert outcome['review_status']=='review_failed' and outcome['receipt_sha256']
        assert outcome['status']=='failed_or_incomplete'
